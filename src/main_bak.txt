use std::{
    env, fs,
    io::{self, Write},
    path::{Path, PathBuf},
    process::Command,
};

/// 支持的压缩文件扩展名
const ARCHIVE_EXTENSIONS: &[&str] = &["part1.rar", "part01.rar", "zip", "rar", "7z"];
/// 图片文件扩展名
const IMAGE_EXTENSIONS: &[&str] = &["png", "jpg", "jpeg"];
/// 重命名前缀
const RENAME_PREFIX: &str = "UC_";

fn main() -> io::Result<()> {
    let args: Vec<String> = env::args().collect();
    let current_dir = if args.len() > 1 {
        PathBuf::from(&args[1])
    } else {
        env::current_dir()?
    };
    let mut temp_password: Option<String>   = None;
    process_directory(&current_dir, &mut temp_password)?;
    Ok(())
}

/// 递归处理目录
fn process_directory(path: &Path, temp_password: &mut Option<String>) -> io::Result<()> {
    // 检查是否存在可执行文件
    if contains_exe_or_html_files(path) {
        return Ok(());
    }

    // 处理当前目录
    loop {
        let archives = find_archives(path);
        if !archives.is_empty() {
            for archive in archives {
                handle_archive(&archive, temp_password)?;
            }
            continue;
        }

        // 检查并重命名图片文件
        if let Some(image_path) = find_single_image(path) {
            let new_path = rename_to_rar(&image_path)?;
            println!("重命名为压缩文件: {:?} -> {:?}", image_path, new_path);
            handle_archive(&new_path, temp_password)?;
            continue;
        }
        break;
    }

    // 递归处理子目录
    for entry in fs::read_dir(path)? {
        let entry = entry?;
        let path = entry.path();
        if path.is_dir() && !contains_exe_or_html_files(&path) {
            let mut new_temp_password = None;
            process_directory(&path, &mut new_temp_password)?;
        }
    }

    Ok(())
}

/// 处理压缩文件
fn handle_archive(archive_path: &Path, temp_password: &mut Option<String>) -> io::Result<()> {
    println!("处理压缩文件: {:?}", archive_path);

    // 检查是否需要密码
    let requires_password = requires_password(archive_path)?;
    let password = if requires_password {
        get_password(archive_path, temp_password)?
    } else {
        None
    };

    // 检查内容是否匹配压缩包名称
    let should_rename = archive_contents_match(archive_path)?;

    let target_path = if should_rename {
        // 重命名压缩包
        let renamed_path = rename_archive(archive_path)?;
        println!("重命名压缩包: {:?} -> {:?}", archive_path, renamed_path);
        renamed_path
    } else {
        archive_path.to_path_buf()
    };

    // 解压文件
    extract_archive(&target_path, password.as_deref())?;
    println!("解压成功: {:?}", target_path);

    // 删除原始压缩包
    if should_rename {
        fs::remove_file(target_path)?;
    } else {
        fs::remove_file(archive_path)?;
    }

    // 保存成功使用的密码
    if requires_password {
        *temp_password = password;
    }

    Ok(())
}

// --- 辅助函数实现 ---

/// 查找目录中的压缩文件
fn find_archives(path: &Path) -> Vec<PathBuf> {
    let mut archives = Vec::new();
    if let Ok(entries) = fs::read_dir(path) {
        for entry in entries.flatten() {
            let path = entry.path();
            if let Some(ext) = path.extension().and_then(|e| e.to_str()) {
                if ARCHIVE_EXTENSIONS.contains(&ext.to_lowercase().as_str()) {
                    archives.push(path);
                }
            }
        }
    }
    archives
}

/// 检查是否包含可执行文件或html文件
fn contains_exe_or_html_files(path: &Path) -> bool {
    if let Ok(entries) = fs::read_dir(path) {
        for entry in entries.flatten() {
            let path = entry.path();
            if let Some(ext) = path.extension().and_then(|e| e.to_str()) {
                if ext.eq_ignore_ascii_case("exe") || ext.eq_ignore_ascii_case("html") {
                    return true;
                }
            }
        }
    }
    false
}

/// 查找单一图片文件
fn find_single_image(path: &Path) -> Option<PathBuf> {
    let mut images = Vec::new();
    if let Ok(entries) = fs::read_dir(path) {
        for entry in entries.flatten() {
            let path = entry.path();
            if let Some(ext) = path.extension().and_then(|e| e.to_str()) {
                if IMAGE_EXTENSIONS.contains(&ext.to_lowercase().as_str()) {
                    images.push(path);
                }
            }
        }
    }
    if images.len() == 1 {
        images.pop()
    } else {
        None
    }
}

/// 检查压缩文件是否需要密码
fn requires_password(archive_path: &Path) -> io::Result<bool> {
    let output = Command::new("7z")
        .arg("l")
        .arg(archive_path)
        .output()?;

    let stdout = String::from_utf8_lossy(&output.stdout);
    Ok(stdout.contains("Encrypted: +"))
}

/// 获取解压密码
fn get_password(
    archive_path: &Path,
    temp_password: &Option<String>,
) -> io::Result<Option<String>> {
    if let Some(pwd) = temp_password {
        if test_password(archive_path, pwd)? {
            return Ok(Some(pwd.clone()));
        }
    }

    loop {
        print!("输入 {} 的解压密码: ", archive_path.display());
        io::stdout().flush()?;
        
        let mut password = String::new();
        io::stdin().read_line(&mut password)?;
        let password = password.trim().to_string();
        
        if test_password(archive_path, &password)? {
            return Ok(Some(password));
        }
        println!("密码错误，请重试");
    }
}

/// 测试密码是否正确
fn test_password(archive_path: &Path, password: &str) -> io::Result<bool> {
    let output = Command::new("7z")
        .arg("t")
        .arg("-p")
        .arg(password)
        .arg(archive_path)
        .output()?;

    Ok(output.status.success())
}

/// 检查压缩包内容是否与文件名匹配
fn archive_contents_match(archive_path: &Path) -> io::Result<bool> {
    let output = Command::new("7z")
        .arg("l")
        .arg(archive_path)
        .output()?;

    let stdout = String::from_utf8_lossy(&output.stdout);
    let base_name = archive_path
        .file_stem()
        .and_then(|n| n.to_str())
        .unwrap_or("");

    // 检查压缩包内文件是否包含与压缩包同名的文件
    Ok(stdout.lines().any(|line| {
        line.contains(base_name) && {
            let parts: Vec<&str> = line.split_whitespace().collect();
            parts.len() > 5 && !parts[5].is_empty()
        }
    }))
}

/// 重命名压缩包
fn rename_archive(archive_path: &Path) -> io::Result<PathBuf> {
    let file_name = archive_path
        .file_name()
        .and_then(|n| n.to_str())
        .unwrap_or_default();
    
    let new_name = format!("{}{}", RENAME_PREFIX, file_name);
    let new_path = archive_path.with_file_name(new_name);
    fs::rename(archive_path, &new_path)?;
    Ok(new_path)
}

/// 重命名图片为压缩文件
fn rename_to_rar(image_path: &Path) -> io::Result<PathBuf> {
    let mut new_path = image_path.to_path_buf();
    new_path.set_extension("rar");
    fs::rename(image_path, &new_path)?;
    Ok(new_path)
}

/// 解压文件
fn extract_archive(archive_path: &Path, password: Option<&str>) -> io::Result<()> {
    let mut command = Command::new("7z");
    command.arg("x").arg(archive_path).arg("-y");

    if let Some(pwd) = password {
        command.arg(format!("-p{}", pwd));
    }

    let status = command.status()?;
    if !status.success() {
        return Err(io::Error::new(
            io::ErrorKind::Other,
            format!("解压失败: {:?}", archive_path),
        ));
    }
    Ok(())
}
